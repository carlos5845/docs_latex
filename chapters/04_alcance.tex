\chapter{Alcance del Plan}
\label{ch:alcance}

El alcance del presente plan abarca formalmente a todos los estamentos, infraestructuras, plataformas y dependencias que conforman el ecosistema tecnológico de la Universidad Nacional del Altiplano.

\section{Comunidad y Personal Universitario}
\begin{itemize}
    \item Personal Docente (ordinarios y contratados).
    \item Personal Administrativo y directivo.
    \item Comunidad Estudiantil de pregrado y posgrado.
    \item Personal de servicios, terceros y contratistas con acceso a recursos institucionales.
\end{itemize}

\section{Infraestructura de Servidores y Data Center}
\begin{itemize}
    \item Servidores físicos y clústeres de virtualización alojados en el Data Center central de la OTI.
    \item Ambientes de producción, pruebas y contingencia en sistemas operativos Linux y Windows Server.
    \item Sistemas de almacenamiento centralizado (NAS / SAN) y nodos de cómputo.
\end{itemize}

\section{Estaciones de Trabajo y Dispositivos Finales}
\begin{itemize}
    \item Computadoras de escritorio y portátiles en oficinas administrativas y decanatos.
    \item Equipos en laboratorios de cómputo de las distintas facultades y escuelas profesionales.
    \item Dispositivos móviles institucionales y acceso remoto de usuarios autorizados.
\end{itemize}

\section{Infraestructura de Comunicaciones y Redes}
\begin{itemize}
    \item Switches de distribución y Switch Core central.
    \item Segmentación lógica de redes mediante VLANs.
    \item Infraestructura de red inalámbrica (Wi-Fi de campus para alumnos, docentes y administrativos).
    \item Enlaces de conectividad WAN e interfaces de conexión hacia la Red Dorsal y proveedores de Internet (ISP).
\end{itemize}

\section{Plataformas, Sistemas y Datos Institucionales}
\begin{itemize}
    \item \textbf{Aplicaciones y portales web institucionales:} Sistema de Matrícula (\path{matricula.unap.edu.pe}), Campus Virtual (\path{campus.unap.edu.pe}), SITRAD (Trámite Documentario), Pasarela de Recaudación y pagos.
    \item \textbf{Bases de datos:} Repositorios centrales de notas, registros de tesorería, planillas de personal y expedientes estudiantiles.
    \item \textbf{Servicios de comunicación institucional:} Servicio de correo electrónico institucional (\path{@unap.edu.pe}), repositorios digitales (DSpace) y servicios en la nube.
    \item \textbf{Proveedores y contratistas:} Servicios tercerizados de desarrollo de software, mantenimiento de telecomunicaciones y soporte de infraestructura.
\end{itemize}
